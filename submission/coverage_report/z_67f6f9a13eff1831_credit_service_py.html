38def reduced_form_summary( 

39 lam: float, 

40 horizons: Sequence[float], 

41 R: float, 

42 r: float = 0.0, 

43) -> dict: 

44 """Build a per-horizon reduced-form credit table under constant hazard. 

45 

46 Args: 

47 lam (float): Constant hazard rate λ (must be ≥ 0). 

48 horizons (Sequence[float]): Time horizons in years at which to 

49 evaluate survival, default density, and spread (e.g. [1,2,3,5]). 

50 R (float): Recovery rate ∈ [0, 1]. 

51 r (float): Risk-free rate for risky ZCB pricing (default 0.0). 

52 

53 Returns: 

54 dict: Summary with keys: 

55 - ``"LGD"`` (float): 1 − R. 

56 - ``"approx_cds"`` (float): §10 landmark (1−R)·λ. 

57 - ``"rows"`` (list[dict]): One dict per horizon with keys 

58 ``t``, ``survival``, ``cum_default``, ``density``, 

59 ``risky_zcb``, ``spread``. ``risky_zcb`` and ``spread`` 

60 are NaN when t = 0. 

61 """ 

62 LGD = 1.0 - R 

63 rows: list[dict] = [] 

64 for t in horizons: 

65 s_t = survival(float(t), lam) 

66 rows.append( 

67 { 

68 "t": float(t), 

69 "survival": s_t, 

70 "cum_default": cumulative_default_prob(float(t), lam), 

71 "density": default_density(float(t), lam), 

72 "risky_zcb": risky_zcb_price(r, float(t), LGD, s_t) if t > 0 else float("nan"), 

73 "spread": credit_spread(float(t), LGD, s_t) if t > 0 else float("nan"), 

74 } 

75 ) 

76 return { 

77 "LGD": LGD, 

78 "approx_cds": cds_par_spread_constant_hazard(lam, R), 

79 "rows": rows, 

80 }

83def interval_default_table( 

84 edges: Sequence[float], 

85 lam: float, 

86) -> list[dict]: 

87 """Build a per-bucket marginal-default table using the constant-hazard formula. 

88 

89 Args: 

90 edges (Sequence[float]): Knot times starting at 0, e.g. [0,1,2,3,5]. 

91 Consecutive pairs define each interval. 

92 lam (float): Constant hazard rate λ (must be ≥ 0). 

93 

94 Returns: 

95 list[dict]: One dict per interval with keys ``"t1"``, ``"t2"``, 

96 and ``"marginal_default"`` = P(t1 < τ ≤ t2). 

97 """ 

98 edges = list(edges) 

99 out: list[dict] = [] 

100 for i in range(len(edges) - 1): 

101 t1, t2 = float(edges[i]), float(edges[i + 1]) 

102 out.append( 

103 { 

104 "t1": t1, 

105 "t2": t2, 

106 "marginal_default": interval_default_prob(t1, t2, lam), 

107 } 

108 ) 

109 return out

114def merton_summary( 

115 V0: float, 

116 B: float, 

117 r: float, 

118 mu: float, 

119 sigma: float, 

120 T: float, 

121) -> dict: 

122 """Full Merton model snapshot under both Q and P measures in one call. 

123 

124 Computes d₁, d₂, and default probability under both risk-neutral (Q, 

125 ν=r) and real-world (P, ν=μ) measures, plus equity and debt values. 

126 

127 Args: 

128 V0 (float): Current firm asset value (must be > 0). 

129 B (float): Debt face value / default barrier (must be > 0). 

130 r (float): Risk-free rate (continuously compounded). 

131 mu (float): Real-world drift of firm assets. 

132 sigma (float): Firm asset volatility (must be > 0). 

133 T (float): Time to maturity in years (must be > 0). 

134 

135 Returns: 

136 dict: Merton snapshot with keys: 

137 - ``"V0"``, ``"B"``, ``"T"`` (float): Input parameters echoed. 

138 - ``"Q"`` (dict): Q-measure results — ``{d1, d2, PD}``. 

139 - ``"P"`` (dict): P-measure results — ``{d1, d2, PD}``. 

140 - ``"E0"`` (float): Equity value (call on assets under Q). 

141 - ``"D0"`` (float): Risky debt value (V0 − E0). 

142 """ 

143 d1_Q, d2_Q = merton_d1_d2(V0, B, r, sigma, T) 

144 d1_P, d2_P = merton_d1_d2(V0, B, mu, sigma, T) 

145 E0 = merton_equity(V0, B, r, sigma, T) 

146 D0 = merton_debt(V0, B, r, sigma, T) 

147 return { 

148 "V0": V0, 

149 "B": B, 

150 "T": T, 

151 "Q": { 

152 "d1": d1_Q, 

153 "d2": d2_Q, 

154 "PD": merton_pd(V0, B, r, sigma, T), 

155 }, 

156 "P": { 

157 "d1": d1_P, 

158 "d2": d2_P, 

159 "PD": merton_pd(V0, B, mu, sigma, T), 

160 }, 

161 "E0": E0, 

162 "D0": D0, 

163 }

196def cds_summary( 

197 lam: float, 

198 R: float, 

199 tenors: Sequence[float], 

200 r: float = 0.03, 

201 premium_freq: float = 1.0, 

202) -> dict: 

203 """Build a CDS summary: constant-hazard approximation and par-spread curve. 

204 

205 Args: 

206 lam (float): Constant hazard rate λ (must be ≥ 0). 

207 R (float): Recovery rate ∈ [0, 1]. 

208 tenors (Sequence[float]): CDS maturities in years for the spread 

209 curve (e.g. [1, 2, 3, 5, 7, 10]). 

210 r (float): Flat risk-free rate for premium-leg discounting 

211 (default 0.03). 

212 premium_freq (float): Premium payments per year (default 1 = annual). 

213 

214 Returns: 

215 dict: CDS summary with keys: 

216 - ``"approx_spread"`` (float): §10 landmark (1−R)·λ. 

217 - ``"curve"`` (list[tuple[float, float]]): Par-spread term 

218 structure as ``[(tenor, spread)]`` pairs (spread in decimal). 

219 """ 

220 approx = cds_par_spread_constant_hazard(lam, R) 

221 curve = cds_spread_curve(list(tenors), lam=lam, r=r, R=R, premium_freq=premium_freq) 

222 return { 

223 "approx_spread": approx, 

224 "curve": curve, # list[(tenor, spread)] 

225 }

257def cva_summary( 

258 exposure_profile: Sequence[float], 

259 marginal_default_probs: Sequence[float], 

260 R: float, 

261 V0: float | None = None, 

262) -> dict: 

263 """Compute CVA from a discrete EPE and marginal-default profile. 

264 

265 Args: 

266 exposure_profile (Sequence[float]): Expected-positive exposure at 

267 each grid point (must be non-negative). 

268 marginal_default_probs (Sequence[float]): Marginal default probability 

269 on each interval (same length; must sum to ≤ 1). 

270 R (float): Recovery rate ∈ [0, 1]. 

271 V0 (float | None): Current portfolio mark-to-market value; if 

272 provided, also returns CVA as a percentage of V0. 

273 

274 Returns: 

275 dict: CVA summary with keys: 

276 - ``"cva"`` (float): CVA in dollars. 

277 - ``"R"`` (float): Recovery rate echoed. 

278 - ``"cva_pct"`` (float): cva / V0 (only present when V0 > 0). 

279 """ 

280 cva = cva_discrete(exposure_profile, marginal_default_probs, R) 

281 out = {"cva": cva, "R": R} 

282 if V0 is not None and V0 > 0: 

283 out["cva_pct"] = cva / V0 

284 return out
