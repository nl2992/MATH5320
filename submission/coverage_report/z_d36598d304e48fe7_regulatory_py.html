31def risk_weighted_assets( 

32 asset_values: Sequence[float], 

33 risk_weights: Sequence[float], 

34) -> float: 

35 """Compute total risk-weighted assets. 

36 

37 RWA = Σ_i w_i · A_i 

38 

39 Args: 

40 asset_values (Sequence[float]): Dollar exposures to each asset. 

41 Lengths must match ``risk_weights``. 

42 risk_weights (Sequence[float]): Corresponding Basel-style risk 

43 weights (typically in [0, 1.5]). Must be non-negative. 

44 

45 Returns: 

46 float: Total risk-weighted assets in dollars. 

47 

48 Raises: 

49 ValueError: If ``asset_values`` and ``risk_weights`` have different 

50 lengths, or if any weight is negative. 

51 """ 

52 a = np.asarray(asset_values, dtype=float) 

53 w = np.asarray(risk_weights, dtype=float) 

54 if len(a) != len(w): 

55 raise ValueError( 

56 f"asset_values and risk_weights must align " 

57 f"(got {len(a)} vs {len(w)})." 

58 ) 

59 if np.any(w < 0): 

60 raise ValueError("risk_weights must be non-negative.") 

61 return float((a * w).sum())

67def capital_ratio(equity: float, rwa: float) -> dict: 

68 """Compute the capital adequacy ratio and Basel pass/fail flag. 

69 

70 ratio = equity / RWA 

71 PASS iff ratio > 0.08 

72 

73 Args: 

74 equity (float): Tier-1 capital in dollars. 

75 rwa (float): Risk-weighted assets in dollars (must be > 0). 

76 

77 Returns: 

78 dict: Capital result with keys: 

79 - ``"ratio"`` (float): equity / rwa. 

80 - ``"pass"`` (bool): True iff ratio > 0.08. 

81 - ``"floor"`` (float): The Basel minimum used (0.08). 

82 

83 Raises: 

84 ValueError: If ``rwa <= 0``. 

85 """ 

86 if rwa <= 0: 

87 raise ValueError(f"rwa must be positive (got {rwa}).") 

88 ratio = equity / rwa 

89 return { 

90 "ratio": float(ratio), 

91 "pass": bool(ratio > CAPITAL_RATIO_FLOOR), 

92 "floor": CAPITAL_RATIO_FLOOR, 

93 }

98def apply_stress_scenario( 

99 portfolio: Portfolio, 

100 current_prices: pd.Series, 

101 shock_map: Mapping[str, float], 

102 pricing_date: date, 

103) -> dict: 

104 """Apply a multiplicative price shock to the portfolio and report PnL. 

105 

106 Args: 

107 portfolio (Portfolio): Stock and option positions to be re-priced. 

108 current_prices (pd.Series): Current spot prices indexed by ticker 

109 symbol. 

110 shock_map (Mapping[str, float]): Per-ticker multiplicative return 

111 shock, e.g. ``{"AAPL": -0.30}`` for a 30% price drop. 

112 Tickers absent from the map are left unshocked. 

113 pricing_date (date): Option pricing date (used by Black-Scholes). 

114 

115 Returns: 

116 dict: Stress-scenario summary with keys: 

117 - ``"V_pre"`` (float): Portfolio value before shock. 

118 - ``"V_post"`` (float): Portfolio value after shock. 

119 - ``"pnl"`` (float): V_post − V_pre (negative = loss). 

120 - ``"pnl_pct"`` (float): pnl / V_pre, or NaN if V_pre == 0. 

121 """ 

122 V_pre = reprice_portfolio(portfolio, current_prices, pricing_date) 

123 

124 shocked = current_prices.copy() 

125 for ticker, shock in shock_map.items(): 

126 if ticker in shocked.index: 

127 shocked[ticker] = shocked[ticker] * (1.0 + shock) 

128 

129 V_post = reprice_portfolio(portfolio, shocked, pricing_date) 

130 pnl = V_post - V_pre 

131 pnl_pct = pnl / V_pre if V_pre != 0 else float("nan") 

132 return { 

133 "V_pre": float(V_pre), 

134 "V_post": float(V_post), 

135 "pnl": float(pnl), 

136 "pnl_pct": float(pnl_pct), 

137 }

163def build_equity_shock_map( 

164 portfolio: Portfolio, 

165 equity_shock: float, 

166) -> dict[str, float]: 

167 """Build a per-ticker multiplicative shock map from a single equity shock. 

168 

169 Applies ``equity_shock`` uniformly to every underlying ticker referenced 

170 by the portfolio (stocks and option underlyings). 

171 

172 Args: 

173 portfolio (Portfolio): Stock and option positions whose underlying 

174 tickers will be enumerated. 

175 equity_shock (float): Multiplicative return shock to apply, e.g. 

176 ``-0.30`` for a 30% price decline. 

177 

178 Returns: 

179 dict[str, float]: Mapping ``{ticker: equity_shock}`` covering all 

180 unique underlying tickers in the portfolio. 

181 """ 

182 tickers: set[str] = set() 

183 for pos in portfolio.stocks: 

184 tickers.add(pos.ticker) 

185 for pos in portfolio.options: 

186 tickers.add(pos.underlying_ticker) 

187 return {t: equity_shock for t in tickers}

212class StressQuarter:

225def project_capital_one_quarter(state: CapitalState, shock: StressQuarter) -> CapitalState: 

226 """Advance the bank's capital position by one stress quarter. 

227 

228 capital_next = capital 

229 + pre_provision_net_revenue 

230 − credit_loss − trading_loss − counterparty_loss 

231 − provisions − dividends − buybacks 

232 rwa_next = rwa + rwa_change 

233 

234 Args: 

235 state (CapitalState): Capital snapshot at the start of the quarter. 

236 shock (StressQuarter): P&L and balance-sheet changes for this quarter. 

237 

238 Returns: 

239 CapitalState: Updated capital snapshot after applying all charges. 

240 

241 Raises: 

242 ValueError: If the resulting RWA is non-positive. 

243 """ 

244 capital_next = ( 

245 state.tier1_capital 

246 + shock.pre_provision_net_revenue 

247 - shock.credit_loss 

248 - shock.trading_loss 

249 - shock.counterparty_loss 

250 - shock.provisions 

251 - shock.dividends 

252 - shock.buybacks 

253 ) 

254 rwa_next = state.rwa + shock.rwa_change 

255 if rwa_next <= 0: 

256 raise ValueError(f"rwa became non-positive after quarter {shock.quarter}.") 

257 return CapitalState(tier1_capital=capital_next, rwa=rwa_next)

260def project_capital_path( 

261 initial_state: CapitalState, 

262 shocks: list[StressQuarter], 

263) -> list[dict]: 

264 """Project capital over multiple stress quarters. 

265 

266 Args: 

267 initial_state (CapitalState): Starting capital and RWA. 

268 shocks (list[StressQuarter]): Ordered list of quarterly shocks; each 

269 element advances the state by one quarter. 

270 

271 Returns: 

272 list[dict]: One dict per quarter with keys ``"quarter"``, 

273 ``"tier1_capital"``, ``"rwa"``, ``"capital_ratio"``. 

274 """ 

275 path: list[dict] = [] 

276 state = initial_state 

277 for shock in shocks: 

278 state = project_capital_one_quarter(state, shock) 

279 ratio = state.tier1_capital / state.rwa 

280 path.append({ 

281 "quarter": shock.quarter, 

282 "tier1_capital": state.tier1_capital, 

283 "rwa": state.rwa, 

284 "capital_ratio": ratio, 

285 }) 

286 return path

384def run_dfast_scenarios( 

385 initial_state: CapitalState, 

386 scenarios: dict[str, list[StressQuarter]] | None = None, 

387 hurdle: float = 0.08, 

388) -> list[dict]: 

389 """Run DFAST-style stress scenarios and return per-scenario summary rows. 

390 

391 Args: 

392 initial_state (CapitalState): Starting Tier-1 capital and RWA. 

393 scenarios (dict[str, list[StressQuarter]] | None): Mapping of 

394 scenario name to a list of quarterly shocks. Defaults to 9 

395 identical quarters built from ``DFAST_ILLUSTRATIVE_SCENARIOS`` 

396 (baseline / adverse / severely_adverse). 

397 hurdle (float): Minimum capital ratio pass threshold (default 0.08). 

398 

399 Returns: 

400 list[dict]: One dict per scenario with keys: 

401 - ``"scenario"`` (str): Scenario name. 

402 - ``"ending_capital"`` (float): Tier-1 capital at quarter 9. 

403 - ``"ending_rwa"`` (float): RWA at quarter 9. 

404 - ``"ending_capital_ratio"`` (float): Ratio at quarter 9. 

405 - ``"min_capital_ratio"`` (float): Worst ratio across all quarters. 

406 - ``"passes"`` (bool): True iff min ratio ≥ hurdle. 

407 """ 

408 if scenarios is None: 

409 scenarios = {} 

410 for name, params in DFAST_ILLUSTRATIVE_SCENARIOS.items(): 

411 quarters = [ 

412 StressQuarter( 

413 quarter=q + 1, 

414 credit_loss=params["credit_loss"], 

415 trading_loss=params["trading_loss"], 

416 counterparty_loss=params["counterparty_loss"], 

417 pre_provision_net_revenue=params["pre_provision_net_revenue"], 

418 provisions=params["provisions"], 

419 rwa_change=params["rwa_change"], 

420 ) 

421 for q in range(9) 

422 ] 

423 scenarios[name] = quarters 

424 

425 results = [] 

426 for scenario_name, quarter_shocks in scenarios.items(): 

427 path = project_capital_path(initial_state, quarter_shocks) 

428 last = path[-1] 

429 results.append({ 

430 "scenario": scenario_name, 

431 "ending_capital": last["tier1_capital"], 

432 "ending_rwa": last["rwa"], 

433 "ending_capital_ratio": last["capital_ratio"], 

434 "min_capital_ratio": min_capital_ratio(path), 

435 "passes": passes_stress(path, hurdle), 

436 }) 

437 return results

442def balance_sheet_equity(assets: float, liabilities: float) -> float: 

443 """Compute book equity using the balance-sheet identity. 

444 

445 Equity = Assets − Liabilities 

446 

447 Args: 

448 assets (float): Total book-value assets (must be ≥ 0). 

449 liabilities (float): Total book-value liabilities (must be ≥ 0). 

450 

451 Returns: 

452 float: Book equity in the same currency units; may be negative for 

453 an insolvent entity. 

454 

455 Raises: 

456 ValueError: If ``assets < 0`` or ``liabilities < 0``. 

457 """ 

458 if assets < 0: 

459 raise ValueError(f"assets must be non-negative (got {assets}).") 

460 if liabilities < 0: 

461 raise ValueError(f"liabilities must be non-negative (got {liabilities}).") 

462 return assets - liabilities

465def balance_sheet_after_asset_loss( 

466 assets: float, liabilities: float, loss: float 

467) -> dict: 

468 """Compute the post-stress balance sheet after an asset write-down. 

469 

470 Assets absorb the loss; liabilities are unchanged (market losses do not 

471 affect the liability side). 

472 

473 Args: 

474 assets (float): Pre-stress total book-value assets (must be ≥ 0). 

475 liabilities (float): Total book-value liabilities (must be ≥ 0). 

476 loss (float): Dollar write-down to deduct from assets (must be ≥ 0). 

477 

478 Returns: 

479 dict: Post-stress balance sheet with keys: 

480 - ``"assets_post"`` (float): assets − loss. 

481 - ``"liabilities"`` (float): unchanged liabilities. 

482 - ``"equity_post"`` (float): assets_post − liabilities. 

483 - ``"solvent"`` (bool): True iff equity_post ≥ 0. 

484 

485 Raises: 

486 ValueError: If ``assets < 0``, ``liabilities < 0``, or ``loss < 0``. 

487 """ 

488 if assets < 0: 

489 raise ValueError(f"assets must be non-negative (got {assets}).") 

490 if liabilities < 0: 

491 raise ValueError(f"liabilities must be non-negative (got {liabilities}).") 

492 if loss < 0: 

493 raise ValueError(f"loss must be non-negative (got {loss}).") 

494 assets_post = assets - loss 

495 equity_post = assets_post - liabilities 

496 return { 

497 "assets_post": assets_post, 

498 "liabilities": liabilities, 

499 "equity_post": equity_post, 

500 "solvent": equity_post >= 0.0, 

501 }
