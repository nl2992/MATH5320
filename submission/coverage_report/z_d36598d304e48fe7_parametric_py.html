1""" 

2parametric.py 

3Delta-Normal (Parametric) VaR and ES. 

4 

5Spec §12: 

6 Exposure vector : x_i = Δ_i S_i 

7 Horizon scaling : μ_h = μ × h, Σ_h = Σ × h 

8 Portfolio mean : m = x' μ_h 

9 Portfolio vol : s = sqrt(x' Σ_h x) 

10 VaR : VaR = -m + s Φ^{-1}(confidence) 

11 ES : ES = -m + s φ(z) / α where α = 1 - confidence 

12""" 

13from __future__ import annotations 

14 

15from datetime import date 

16 

17import numpy as np 

18import pandas as pd 

19from scipy.stats import norm 

20 

21from src.portfolio.portfolio import portfolio_exposure 

22from src.risk.estimators import get_mean_cov, manual_mean_cov 

23from src.risk.returns import compute_log_returns 

24from src.schemas import Portfolio 

25 

26 

27def parametric_var_es( 

28 portfolio: Portfolio, 

29 prices: pd.DataFrame, 

30 pricing_date: date, 

31 lookback_days: int, 

32 horizon_days: int, 

33 var_confidence: float, 

34 es_confidence: float, 

35 estimator: str = "window", 

36 ewma_N: int = 60, 

37 calibration_mode: str = "historical", 

38 manual_market_params: dict | None = None, 

39) -> dict: 

40 """Parametric (Delta-Normal) VaR and ES. 

41 

42 The delta-normal method approximates the portfolio P&L as a linear function 

43 of underlying log returns. Given the exposure vector x (dollar-delta per 

44 underlying) and scaled distribution N(μ_h, Σ_h): 

45 m = x' μ_h, s = sqrt(x' Σ_h x) 

46 VaR = -m + s × Φ⁻¹(var_confidence) 

47 ES = -m + s × φ(z_es) / (1 − es_confidence) 

48 

49 Args: 

50 portfolio (Portfolio): Stock and option positions. Options contribute 

51 via their Black-Scholes delta-dollar exposure only (no repricing). 

52 prices (pd.DataFrame): Aligned price history (DatetimeIndex × tickers). 

53 pricing_date (date): Valuation date for computing option deltas. 

54 lookback_days (int): Trailing observations for mean/cov estimation. 

55 horizon_days (int): Risk horizon h; μ_h = μ×h, Σ_h = Σ×h. 

56 var_confidence (float): VaR confidence level, e.g. 0.99. 

57 es_confidence (float): ES averaging confidence, e.g. 0.975. 

58 estimator (str): ``"window"`` (simple rolling) or ``"ewma"`` (exponentially 

59 weighted). Default ``"window"``. 

60 ewma_N (int): EWMA half-life parameter N; λ = (N-1)/(N+1). Only used 

61 when estimator=``"ewma"``. 

62 calibration_mode (str): ``"historical"`` (estimate from prices) or 

63 ``"manual"`` (supply μ and Σ directly via manual_market_params). 

64 manual_market_params (dict | None): Required when calibration_mode=``"manual"``. 

65 Must contain ``"mu_daily"`` (pd.Series) and ``"cov_daily"`` (pd.DataFrame) 

66 keyed by underlying ticker. 

67 

68 Returns: 

69 dict: Result dictionary with keys: 

70 - ``"var"`` (float): Parametric VaR in dollars. 

71 - ``"es"`` (float): Parametric ES in dollars. 

72 - ``"var_confidence"`` (float): var_confidence passed in. 

73 - ``"es_confidence"`` (float): es_confidence passed in. 

74 - ``"portfolio_mean"`` (float): m = x' μ_h. 

75 - ``"portfolio_vol"`` (float): s = sqrt(x' Σ_h x). 

76 - ``"exposure"`` (pd.Series): Dollar-delta exposure vector x. 

77 - ``"calibration_mode"`` (str): Mode used. 

78 """ 

79 # Current spots 

80 spots_0 = prices.iloc[-1] 

81 

82 # Exposure vector x (dollar-delta per underlying) 

83 exposure = portfolio_exposure(portfolio, spots_0, pricing_date) 

84 

85 # Restrict to underlyings that have price data 

86 underlyings = [u for u in exposure.index if u in prices.columns] 

87 exposure = exposure[underlyings] 

88 

89 # Estimate daily mean and covariance from log returns 

90 if calibration_mode == "manual": 

91 mu_daily, cov_daily = manual_mean_cov(manual_market_params or {}, underlyings) 

92 else: 

93 log_ret = compute_log_returns(prices[underlyings]) 

94 mu_daily, cov_daily = get_mean_cov(log_ret, lookback_days, estimator, ewma_N) 

95 

96 # Horizon scaling 

97 mu_h = mu_daily * horizon_days 

98 cov_h = cov_daily * horizon_days 

99 

100 x = exposure.values # shape (k,) 

101 mu_h_arr = mu_h.values 

102 cov_h_arr = cov_h.values 

103 

104 # Portfolio-level mean and volatility 

105 m = float(x @ mu_h_arr) 

106 variance = float(x @ cov_h_arr @ x) 

107 s = float(np.sqrt(max(variance, 0.0))) 

108 

109 # VaR 

110 z_var = norm.ppf(var_confidence) 

111 var = float(-m + s * z_var) 

112 

113 # ES — use es_confidence separately 

114 z_es = norm.ppf(es_confidence) 

115 alpha_es = 1.0 - es_confidence 

116 es = float(-m + s * norm.pdf(z_es) / alpha_es) 

117 

118 return { 

119 "var": var, 

120 "es": es, 

121 "var_confidence": var_confidence, 

122 "es_confidence": es_confidence, 

123 "portfolio_mean": m, 

124 "portfolio_vol": s, 

125 "exposure": exposure, 

126 "calibration_mode": calibration_mode, 

127 }
