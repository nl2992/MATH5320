def ewma_mean_var(series, lam):
    """
    Exact EWMA mean and variance via recurrence (Lecture 4, Slide 48).
    m_N = (1-λ)·a_N  + λ·m_{N-1}    ← mean recurrence
    r_N = (1-λ)·a_N² + λ·r_{N-1}    ← 2nd moment recurrence
    v_N = r_N - m_N²                 ← variance
    """
    alpha = 1 - lam
    a = series.values
    m = np.zeros(len(a))
    r = np.zeros(len(a))
    m[0] = a[0]
    r[0] = a[0] ** 2
    for i in range(1, len(a)):
        m[i] = alpha * a[i]     + lam * m[i-1]   
        r[i] = alpha * a[i]**2  + lam * r[i-1]   
    v = r - m**2                                   
    return (
        pd.Series(m * TRADING_DAYS, index=series.index),  
        pd.Series(v * TRADING_DAYS, index=series.index),   
    )

# Breakpoints and hazard values
# λ(t) = 0.010  for t ≤ 1.0
#        0.011  for 1.0 < t ≤ 2.0
#        0.012  for t > 2.0
def hazard(t):
    """Piecewise constant λ(t)."""
    t = np.atleast_1d(np.array(t, dtype=float))
    lam = np.where(t <= 1.0, 0.010,
          np.where(t <= 2.0, 0.011, 0.012))
    return lam
def cumulative_hazard(t):
    """
    Λ(t) = ∫₀ᵗ λ(u) du  — integrated piecewise.
    Piece 1 (t ≤ 1):     Λ(t) = 0.010·t
    Piece 2 (1 < t ≤ 2): Λ(t) = 0.010·1 + 0.011·(t-1)
    Piece 3 (t > 2):     Λ(t) = 0.010·1 + 0.011·1 + 0.012·(t-2)
    """
    t = np.atleast_1d(np.array(t, dtype=float))
    L = np.where(
        t <= 1.0,
        0.010 * t,
        np.where(
            t <= 2.0,
            0.010 * 1 + 0.011 * (t - 1.0),
            0.010 * 1 + 0.011 * 1 + 0.012 * (t - 2.0)
        )
