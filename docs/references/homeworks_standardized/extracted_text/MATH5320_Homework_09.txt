def short_var_gbm(mu, sigma, h, V0, alpha):
    """99% VaR for a SHORT position under GBM."""
    z = norm.ppf(alpha)
    return V0 * (np.exp(mu * h + z * sigma * np.sqrt(h)) - 1)

def compute_var_es(losses, var_level=0.99, es_level=0.975):
    var_value = np.quantile(losses, var_level)
    es_cutoff = np.quantile(losses, es_level)
    es_value  = losses[losses >= es_cutoff].mean()
    return var_value, es_value
def mc_short_underlying(
    aapl_returns_window, cat_returns_window,
    s_aapl_t, s_cat_t, weight_aapl_t, weight_cat_t,
    position_value=10_000, horizon_days=5, n_sims=10_000
):
    """MC VaR/ES for a SHORT portfolio position, simulating the two under
    X = np.column_stack([aapl_returns_window.values, cat_returns_window.v
    mu  = X.mean(axis=0)
    cov = np.cov(X.T, ddof=1)
    mu_h  = mu * horizon_days
    cov_h = cov * horizon_days
    sim_r = np.random.multivariate_normal(mu_h, cov_h, size=n_sims)
    sim_aapl = s_aapl_t * np.exp(sim_r[:, 0])
    sim_cat  = s_cat_t  * np.exp(sim_r[:, 1])
    # Number of shares for a $10k SHORT position
    n_aapl = (position_value * weight_aapl_t) / s_aapl_t
    n_cat  = (position_value * weight_cat_t)  / s_cat_t
    # Short portfolio value at horizon
    sim_port_value = n_aapl * sim_aapl + n_cat * sim_cat
    # SHORT loss = portfolio appreciated → we lose
    losses = sim_port_value - position_value
    var99, es975 = compute_var_es(losses, 0.99, 0.975)
    return var99, es975

Formula P vs MC(A,C) comparison
The formula approach treats P as a single GBM, estimating  and  directly from P's
log returns. The MC approach simulates the two underlyings as correlated GBMs and
reconstructs P.
The MC approach captures the nonlinear composition of P from A and C (since
portfolio weights drift over time), whereas the formula assumes P itself is exactly
lognormal. Since the portfolio is a weighted sum of lognormals (not a product), P is
not exactly lognormal — the MC is more accurate, though the formula is a reasonable
approximation when correlations and weight drift are moderate.
